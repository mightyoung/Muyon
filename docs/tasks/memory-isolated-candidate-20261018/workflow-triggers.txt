FILE .github/workflows/package-artifacts.yml
# Manual internal packages only. Do not dispatch before storage budget approval.
name: package-artifacts
run-name: Package ${{ inputs.platform }} at ${{ inputs.target_sha }}

on:
  workflow_dispatch:
    inputs:
      target_sha:
        description: 'Reviewed source commit (full lowercase 40-character SHA)'
        required: true
        type: string
      platform:
        description: 'One platform only (internal validation, not a release)'
        required: true
        type: choice
        options:
          - android
          - macos
          - windows
