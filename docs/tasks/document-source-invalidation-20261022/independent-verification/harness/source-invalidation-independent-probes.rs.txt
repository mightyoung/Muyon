fn independent_grant_target_replacement_invalidates_old_and_preserves_independent_new_source() {
 let (_f,store,source,local,kind)=fixture();
 let blob=upload(&store,"project",b"Another original synthetic evidence source.");
 let metadata=store.register_document_trusted("project","another","Another",DocumentFormat::Text,&blob).unwrap();
 let other=DocumentAccess{document:metadata.document,authority:AuthorityBinding{grant_id:"independent-other-grant".into(),generation:Revision::new(1),owner_epoch:"epoch".into()}};
 store.issue_document_grant_trusted(&other.document,&other.authority,&[DocumentOperation::Inspect,DocumentOperation::ReadText],expiry()).unwrap();
 let evidence=StoreEvidence::new(&store);let processing=evidence.process(&other,InputKind::Utf8Text).unwrap();let hits=evidence.find_literal(&other,&processing,"synthetic",1).unwrap();
 let proof=store.resolve_local_evidence_trusted(&local.project,ArtifactProjectOperation::LocalDraft,&other,&processing,&hits[0]).unwrap();
 let otherlocal=LocalAccess{object:LocalObjectRef{object_id:"other-object".into(),target:proof.target(),..local.object.clone()},..local.clone()};
 store.issue_local_evidence_binding_trusted(&otherlocal,&kind,expiry(),&proof).unwrap();drop(proof);
 let next=AuthorityBinding{generation:source.authority.generation.next().unwrap(),..source.authority.clone()};
 store.issue_document_grant_trusted(&other.document,&next,&[DocumentOperation::Inspect,DocumentOperation::ReadText],expiry()).unwrap();

 let events=store.pending_document_source_notifications_trusted(None,256).unwrap();assert_eq!(events.len(),1);assert_eq!(events[0].object_id,"object");
}
#[test]
fn independent_outbox_commit_rollback_and_stale_ack_preserve_later_generation() {
 let (f,store,mut source,mut local,kind)=fixture();
 let metadata=store.read_document(&source).unwrap();let observer=Db::open(f.path().to_str().unwrap(),true,false).unwrap();
 let tx=store.database().transaction().unwrap();store.document_clone_in_transaction(&tx,&source.document,"rollback",&metadata.blob).unwrap();
 assert!(store.pending_document_source_notifications_trusted(None,256).is_err());
 assert!(observer.query("SELECT 1 FROM document_source_notifications_v1 WHERE pending=1",&[]).unwrap().is_empty());drop(tx);
 assert!(store.pending_document_source_notifications_trusted(None,256).unwrap().is_empty());assert_eq!(active(&store),"1");
 store.revoke_document_grant_trusted(&source.authority).unwrap();let first=store.pending_document_source_notifications_trusted(None,256).unwrap();assert_eq!(first.len(),1);
 assert_eq!(store.ack_document_source_notifications_trusted(&first).unwrap(),1);assert_eq!(store.ack_document_source_notifications_trusted(&first).unwrap(),0);
 source.authority.generation=source.authority.generation.next().unwrap();store.issue_document_grant_trusted(&source.document,&source.authority,&[DocumentOperation::Inspect,DocumentOperation::ReadText],expiry()).unwrap();
 local.binding_generation=local.binding_generation.next().unwrap();bind(&store,&source,&local,&kind);store.revoke_document_grant_trusted(&source.authority).unwrap();
 let second=store.pending_document_source_notifications_trusted(None,256).unwrap();assert!(second[0].generation.value()>first[0].generation.value());
 assert_eq!(store.ack_document_source_notifications_trusted(&first).unwrap(),0);assert_eq!(store.pending_document_source_notifications_trusted(None,256).unwrap(),second);
}
