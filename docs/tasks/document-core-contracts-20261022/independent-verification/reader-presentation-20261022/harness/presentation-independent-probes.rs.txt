
use muyon_core::documents::reader::{DocumentReaderHost, PresentationPhase};
fn independent_presentation_fixture(deadline: u64) -> (Fixture, DocumentReaderHost, DocumentAccess) {
 let f=Fixture::new(); let store=f.initialize(); let m=register(&store); let authority=binding();
 store.issue_document_grant_trusted(&m.document,&authority,&[DocumentOperation::Render],deadline).unwrap();
 let access=DocumentAccess{document:m.document,authority}; (f,DocumentReaderHost::new(store),access)
}
fn independent_pins(f:&Fixture)->usize{ f.db().query("SELECT pin_id FROM blob_pins WHERE kind='reader'",&[]).unwrap().len() }
#[test]
fn independent_old_ticket_cannot_read_check_advance_or_close_current_frame() {
 let (f,mut host,a)=independent_presentation_fixture(expires());
 let old=host.open_presentation(&a).unwrap();
 assert_eq!(host.read_presentation_chunk(&old.status.ticket,65536).unwrap().bytes,PDF);
 let next=host.advance_presentation_frame(&old.status.ticket).unwrap();
 assert!(host.read_presentation_chunk(&old.status.ticket,1).is_err());
 assert!(host.check_presentation(&old.status.ticket).is_err());
 assert!(host.advance_presentation_frame(&old.status.ticket).is_err());
 assert!(host.close_presentation(&old.status.ticket).is_err());
 assert_eq!(host.check_presentation(&next.ticket).unwrap().phase,PresentationPhase::Ready);
 assert_eq!(independent_pins(&f),1);
 let replacement=host.open_presentation(&a).unwrap();
 assert_ne!(replacement.status.ticket.session_id(),next.ticket.session_id());
 host.close_presentation(&next.ticket).unwrap();
 assert!(host.check_presentation(&replacement.status.ticket).is_ok());
 host.close_all().unwrap(); assert_eq!(independent_pins(&f),0);
}
#[test]
fn independent_short_grant_deadline_survives_eof_frame_and_status_refresh() {
 let deadline=SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_millis() as u64+1200;
 let (f,mut host,a)=independent_presentation_fixture(deadline);
 let opened=host.open_presentation(&a).unwrap();
 assert!(opened.status.valid_for_ms>0 && opened.status.valid_for_ms<=1200);
 assert!(host.read_presentation_chunk(&opened.status.ticket,65536).unwrap().eof);
 let next=host.advance_presentation_frame(&opened.status.ticket).unwrap();
 let before=host.check_presentation(&next.ticket).unwrap();
 std::thread::sleep(std::time::Duration::from_millis(35));
 let after=host.check_presentation(&next.ticket).unwrap();
 assert!(after.valid_for_ms<before.valid_for_ms);
 assert_eq!(f.db().query("SELECT expires_at_ms FROM document_grants",&[]).unwrap()[0][0].as_deref(),Some(deadline.to_string().as_str()));
 let now=SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_millis() as u64;
 std::thread::sleep(std::time::Duration::from_millis(deadline.saturating_sub(now)+30));
 assert!(host.check_presentation(&next.ticket).is_err());
 assert!(host.read_presentation_chunk(&next.ticket,1).is_err());
 assert_eq!(independent_pins(&f),0);
 assert!(!host.take_presentation_invalidations().invalidated.is_empty());
}
#[test]
fn independent_notification_overflow_requires_clear_all_and_drain_resets() {
 let (f,mut host,a)=independent_presentation_fixture(expires());
 let mut last=None;
 for _ in 0..26 {last=Some(host.open_presentation(&a).unwrap());assert_eq!(independent_pins(&f),1);}
 let events=host.take_presentation_invalidations(); assert!(events.clear_all); assert!(events.invalidated.len()<=8);
 let drained=host.take_presentation_invalidations(); assert!(!drained.clear_all && drained.invalidated.is_empty());
 let last=last.unwrap(); assert!(host.check_presentation(&last.status.ticket).is_ok());
 host.close_all().unwrap();assert_eq!(independent_pins(&f),0);
 assert_eq!(host.take_presentation_invalidations().invalidated.len(),1);
}
#[test]
fn independent_legacy_entrypoints_cannot_read_validate_or_close_presentation() {
 let (f,mut host,a)=independent_presentation_fixture(expires());let opened=host.open_presentation(&a).unwrap();
 for ready in [false,true] {
  assert!(host.read_chunk(&opened.session.session_id,65536).is_err());
  assert!(host.is_current(&opened.session.session_id).is_err());
  assert!(host.close(&opened.session.session_id).is_err());
  assert_eq!(independent_pins(&f),1);
  let status=host.check_presentation(&opened.status.ticket).unwrap();
  assert_eq!(status.phase,if ready {PresentationPhase::Ready}else{PresentationPhase::Loading});
  if !ready {assert!(host.read_presentation_chunk(&opened.status.ticket,65536).unwrap().eof);}
 }
 host.close_presentation(&opened.status.ticket).unwrap();assert_eq!(independent_pins(&f),0);
}
