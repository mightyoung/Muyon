fn archive(rev: i64) -> Vec<u8> {
    let mut b = b"MYBD\0\0\0\x01".to_vec();
    b.extend(rev.to_be_bytes());
    for _ in 0..3 {
        b.extend([0u8; 6]);
    }
    b.push(0);
    let hash = Sha256::digest(&b);
    b.extend_from_slice(&hash);
    b
}
fn binding(v: &Value) -> DraftBinding {
    DraftBinding {
        namespace_id: v["namespace_id"].as_str().unwrap().into(),
        draft_id: v["draft_id"].as_str().unwrap().into(),
        expected_generation: v["registry_generation"].as_str().unwrap().into(),
        expected_stored_revision: v["stored_revision"].as_str().unwrap().into(),
    }
}
// Existing storage/CAS regressions now exercise the required prepare-before-save protocol.
// New journal regressions below call the raw commit API to test refusal explicitly.
fn save(
    store: &BusinessStore,
    conversation: &str,
    project: Option<&str>,
    binding: &DraftBinding,
    attempt: &str,
    memory: &str,
    archive: &[u8],
) -> Result<Value, &'static str> {
    prepare_save(
        store,
        conversation,
        project,
        binding,
        attempt,
        SaveRevisions {
            exported_memory_revision: memory,
            host_editor_revision: memory,
        },
        archive,
    )?;
    muyon_core::bot_drafts::save(
        store,
        conversation,
        project,
        binding,
        attempt,
        memory,
        archive,
    )
}

fn original_receipt_is_atomic_idempotent_and_survives_callback_loss() {
    let f = Fixture::new();
    let s = f.open();

    let b = binding(&d);

        lookup_save(&s, C, None, &b.namespace_id, &b.draft_id, A).unwrap(),

#[test]
fn archive_storage_null_oversize_blob_and_invalid_text_refuse() {
    let f = Fixture::new();
    let s = f.open();
    let b = binding(&resolve(&s, C, None).unwrap());
    save(&s, C, None, &b, A, "0", &archive(1)).unwrap();
    let mut next = b.clone();
    next.expected_stored_revision = "1".into();
    let good: String = archive(1).iter().map(|v| format!("{v:02x}")).collect();
    // A well-formed archive stored as BLOB must not become text implicitly.
    f.db()
        .exec(
            "UPDATE local_bot_drafts SET archive_hex=CAST(? AS BLOB)",
            &[&good],
        )
        .unwrap();
    assert_eq!(read(&s, C, None, &next).unwrap_err(), "local_draft_corrupt");
    for bad in [
        "0".repeat(65536),
        "0".repeat(65537),
        "0".repeat(65538),
        String::new(),
    ] {
        f.db()
            .exec("UPDATE local_bot_drafts SET archive_hex=?", &[&bad])
            .unwrap();
        assert_eq!(read(&s, C, None, &next).unwrap_err(), "local_draft_corrupt");
    }
    f.db()
        .exec(
            "UPDATE local_bot_drafts SET archive_hex=? || char(0) || ?",
            &[&good, &"0".repeat(65536)],
        )
        .unwrap();
    assert_eq!(read(&s, C, None, &next).unwrap_err(), "local_draft_corrupt");
    f.db()
        .exec("UPDATE local_bot_drafts SET archive_hex=42", &[])
        .unwrap();
    assert_eq!(read(&s, C, None, &next).unwrap_err(), "local_draft_corrupt");
    f.db()
        .exec(
            "UPDATE local_bot_drafts SET archive_hex=?,archive_digest=CAST(archive_digest AS BLOB)",
            &[&good],
        )
        .unwrap();
    assert_eq!(read(&s, C, None, &next).unwrap_err(), "local_draft_corrupt");
    f.db()
        .exec(
            "UPDATE local_bot_drafts SET archive_digest=CAST(archive_digest AS TEXT)",
            &[],
        )
        .unwrap();
    assert_eq!(read(&s, C, None, &next).unwrap()["status"], "stored");
    f.db()
        .exec(
            "UPDATE local_bot_drafts SET archive_hex=NULL,archive_digest=NULL",
            &[],
        )
        .unwrap();
    assert_eq!(read(&s, C, None, &next).unwrap_err(), "local_draft_corrupt");
}

#[test]
fn large_valid_canonical_archive_roundtrips_without_projection_truncation() {
    let f = Fixture::new();
    let s = f.open();
    let b = binding(&resolve(&s, C, None).unwrap());
    let mut payload = b"MYBD\0\0\0\x01".to_vec();
    payload.extend(1i64.to_be_bytes());
    for units in [60, 8000, 160] {
        let text = "中".repeat(units);
        payload.extend([0u8, 1u8]); // Canonical nonempty USER field, locked.
        payload.extend((text.len() as i32).to_be_bytes());
        payload.extend(text.as_bytes());
    }
    payload.push(0); // No template.
    let hash = Sha256::digest(&payload);
    payload.extend_from_slice(&hash);
    assert!(payload.len() < 32768);
    save(&s, C, None, &b, A, "0", &payload).unwrap();
    let mut next = b;
    next.expected_stored_revision = "1".into();
    assert_eq!(
        read(&s, C, None, &next).unwrap()["archive_hex"]
            .as_str()
            .unwrap()
            .len(),
        payload.len() * 2
    );
}

    let f = Fixture::new();
    let s = f.open();
    let first = resolve(&s, C, None).unwrap();
    assert!(first["save_attempt"].is_null());
    let b = binding(&first);
    assert_eq!(
        muyon_core::bot_drafts::save(&s, C, None, &b, A, "17", &archive(1)).unwrap_err(),
        "local_draft_not_prepared"
    );
    let prepared = prepare_save(
        &s,
        C,
        None,
        &b,
        A,
        SaveRevisions {
            exported_memory_revision: "17",
            host_editor_revision: "42",
        },
        &archive(1),
    )
    .unwrap();
    assert_eq!(prepared["state"], "prepared");
    assert_eq!(prepared["host_editor_revision"], "42");
    assert_eq!(prepared["next_stored_revision"], "1");
    drop(s);
    let s = f.open();
    assert_eq!(resolve(&s, C, None).unwrap()["save_attempt"], prepared);
    assert_eq!(read(&s, C, None, &b).unwrap()["status"], "not_saved");
    assert_eq!(
        prepare_save(
            &s,

            None,
            &b,
            B,

    assert_eq!(terminal["state"], "not_committed");

        muyon_core::bot_drafts::save(&s, C, None, &b, A, "17", &archive(1)).unwrap_err(),
        "local_draft_terminal"
    );
    assert_eq!(

        terminal
    );

        &s,
        C,
        None,

            exported_memory_revision: "18",
            host_editor_revision: "43",
        },

#[test]
fn independent_lookup_save_race_has_one_consistent_terminal_outcome() {
    use std::sync::{Arc,Mutex,Barrier};
    let mut committed=0;let mut fenced=0;
    for _ in 0..32 {
        let f=Fixture::new();let s=f.open();let b=binding(&resolve(&s,C,None).unwrap());
        prepare_save(&s,C,None,&b,A,SaveRevisions{exported_memory_revision:"17",host_editor_revision:"42"},&archive(1)).unwrap();
        let shared=Arc::new(Mutex::new(s));let gate=Arc::new(Barrier::new(3));
        let save_store=shared.clone();let save_gate=gate.clone();let save_binding=b.clone();
        let save_thread=std::thread::spawn(move||{save_gate.wait();muyon_core::bot_drafts::save(&save_store.lock().unwrap(),C,None,&save_binding,A,"17",&archive(1))});
        let lookup_store=shared.clone();let lookup_gate=gate.clone();let lookup_binding=b.clone();
        let lookup_thread=std::thread::spawn(move||{lookup_gate.wait();lookup_save(&lookup_store.lock().unwrap(),C,None,&lookup_binding.namespace_id,&lookup_binding.draft_id,A)});
        gate.wait();let saved=save_thread.join().unwrap();let outcome=lookup_thread.join().unwrap().unwrap();let s=shared.lock().unwrap();
        match outcome["status"].as_str().unwrap() {
            "committed_local_storage"=>{committed+=1;assert_eq!(saved.unwrap(),outcome);assert_eq!(resolve(&s,C,None).unwrap()["stored_revision"],"1");},
            "not_committed_local_storage"=>{fenced+=1;assert_eq!(saved.unwrap_err(),"local_draft_terminal");assert_eq!(resolve(&s,C,None).unwrap()["stored_revision"],"0");assert_eq!(muyon_core::bot_drafts::save(&s,C,None,&b,A,"17",&archive(1)).unwrap_err(),"local_draft_terminal");},
            other=>panic!("unexpected terminal {other}"),
        }
        assert_eq!(lookup_save(&s,C,None,&b.namespace_id,&b.draft_id,A).unwrap(),outcome);
    }
    assert_eq!(committed+fenced,32);println!("observed32 mutex races: committed={committed}, fenced={fenced}; finite observation, not exhaustive interleaving proof");
}
