fn expiry() -> u64 {
    std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .unwrap()
        .as_millis() as u64
        + 25_000
}
fn upload(store: &BusinessStore, project: &str, bytes: &[u8]) -> crate::blobs::FinishedBlob {
    let upload = store.begin_upload(project).unwrap();
    store.append_chunk(&upload.upload_id, bytes).unwrap();
    store
        .finish_blob(&upload.upload_id, BlobKind::Document)
        .unwrap()
}
fn bind(store: &BusinessStore, source: &DocumentAccess, local: &LocalAccess, kind: &LocalKind) {
    let evidence = StoreEvidence::new(store);
    let processing = evidence.process(source, InputKind::Utf8Text).unwrap();
    let hits = evidence
        .find_literal(source, &processing, "synthetic", 1)
        .unwrap();
    let proof = store
        .resolve_local_evidence_trusted(
            &local.project,
            ArtifactProjectOperation::LocalDraft,
            source,
            &processing,
            &hits[0],
        )
        .unwrap();
    assert_eq!(local.object.target, proof.target());
    store
        .issue_local_evidence_binding_trusted(local, kind, expiry(), &proof)
        .unwrap();
}

#[test]
fn independent_notification_trigger_must_not_restore_revoked_grant() {
 let (_f,store,source,_,_)=fixture();
 store.database().exec("CREATE TRIGGER independent_restore AFTER INSERT ON document_source_notifications_v1 BEGIN UPDATE document_grants SET active=1 WHERE grant_id='source-grant'; END",&[]).unwrap();
 let result=store.revoke_document_grant_trusted(&source.authority);
 let granted=store.database().query("SELECT active FROM document_grants WHERE grant_id='source-grant'",&[]).unwrap();
 let queue=store.pending_document_source_notifications_trusted(None,256).unwrap();
 println!("revocation_ok={} grant_active={:?} source_active={} queue_pending={} authorized_read_ok={}",result.is_ok(),granted,active(&store),queue.len(),store.read_document(&source).is_ok());
 assert!(result.is_err(),"must reject and rollback when queue trigger restored revoked grant before commit");
}
