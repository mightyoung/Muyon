fn expiry() -> u64 { std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).unwrap().as_millis() as u64 + 25_000 }
fn upload(store: &BusinessStore, project: &str, bytes: &[u8]) -> crate::blobs::FinishedBlob {
    let upload = store.begin_upload(project).unwrap();
    store.append_chunk(&upload.upload_id, bytes).unwrap();
    store.finish_blob(&upload.upload_id, BlobKind::Document).unwrap()
}
fn bind(store: &BusinessStore, source: &DocumentAccess, local: &LocalAccess, kind: &LocalKind) {
    let evidence = StoreEvidence::new(store);
    let processing = evidence.process(source, InputKind::Utf8Text).unwrap();
    let hits = evidence.find_literal(source, &processing, "synthetic", 1).unwrap();
    let proof = store.resolve_local_evidence_trusted(&local.project, ArtifactProjectOperation::LocalDraft, source, &processing, &hits[0]).unwrap();
    assert_eq!(local.object.target, proof.target());
    store.issue_local_evidence_binding_trusted(local, kind, expiry(), &proof).unwrap();
}
